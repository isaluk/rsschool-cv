Codewars Kata: Advanced Events

function Event() {

  let events = [];

  this.subscribe = function (...args) {
    args.forEach((a) => {
      if (typeof a === "function") {
        events.push(a);
      }
    });
  };

  this.unsubscribe = function (...args) {
    args.forEach((a) => {
      if (typeof a === "function") {
        let ind = events.lastIndexOf(a);

        if (ind !== -1) {
          events.splice(ind, 1);
        }
      }
    });
  };

  this.emit = function (...args) {
    const n = [...events];
    n.forEach((a) => a.call(this, ...args));
  };

}
